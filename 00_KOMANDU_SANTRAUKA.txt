FAILAI IR KATALOGAI / FILES & DIRS
  pwd                 kur esu / current dir
  ls -la              sąrašas su teisėmis ir paslėptais / list, long, hidden
  cd kat ; cd .. ; cd ~ ; cd -      eiti / go (up, home, back)
  mkdir -p a/b        kurti (su tėvais) / create (with parents)
  touch f             tuščias failas / empty file
  cp -r src dst       kopijuoti (-r katalogus) / copy
  mv a b              perkelti / pervadinti / move or rename
  rm -rf kat          trinti / delete (recursive, force)
  ln -s tikslas nuoroda   simbolinė nuoroda / symlink
  cat f ; less f ; head -n 5 f ; tail -n 5 f ; tail -f log
  grep -n žodis f ; grep -r žodis kat ; grep -i (nepaisyti dydžio)
  wc -l -w -m -c f    eilutės žodžiai simboliai baitai / lines words chars bytes
  sort -u f ; uniq ; nl f ; cat -n f
  find kat -name "*.txt" -type f 2>/dev/null ; locate f
  stat -c %s f        dydis baitais / size in bytes
  file f              failo tipas / file type
  echo "t" > f (perrašyti) ; >> f (pridėti) ; cmd | cmd (pipe) ; tee f
  sed -i 's/a/b/g' f  pakeisti tekste / replace in file

VARTOTOJAI / USERS
  sudo useradd -m -d /home/x -g users -G sudo,audio -s /bin/bash vardas
  sudo passwd vardas ; echo "v:slapt" | sudo chpasswd
  sudo usermod -aG grupė vardas (pridėti) ; sudo gpasswd -d vardas grupė (pašalinti)
  sudo userdel -r vardas ; sudo groupadd g ; sudo groupdel g
  id ; groups ; whoami ; who ; w ; su - vardas ; sudo -i ; exit
  /etc/passwd /etc/shadow /etc/group /etc/sudoers (visudo)

TEISĖS / PERMISSIONS   r=4 w=2 x=1   u g o   -rwxr-xr-x = 755
  chmod 644 f ; chmod u+x f ; chmod -R 755 kat ; chmod o= f
  sudo chown vart:grupė f ; sudo chgrp grupė f ; umask
  setfacl -m u:vardas:rw f ; getfacl f

PROCESAI / PROCESSES
  ps aux ; ps -ef ; ps -u vardas ; ps aux --sort=-%mem | head
  pstree -p ; top (M atmintis, P cpu, k kill, q) ; htop
  pgrep -a vardas ; pidof ; kill PID ; kill -9 PID ; pkill vardas ; killall vardas
  cmd & ; jobs ; fg ; bg ; Ctrl+Z ; nohup cmd & ; timeout 10 cmd
  ps aux | awk '$8 ~ /Z/'   zombiai

TARNYBOS / SERVICES (systemd)
  systemctl status|start|stop|restart|reload|enable|disable X
  systemctl enable --now X ; systemctl is-active X ; systemctl --failed
  systemctl list-units --type=service [--all] [--state=inactive]
  journalctl -u X [-b] [-f] [-p err] [--since "1 hour ago"]
